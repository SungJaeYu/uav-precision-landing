cmake_minimum_required(VERSION 3.20)
project(uav_precision_landing LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LANDING_SANITIZERS "Enable address and undefined behavior checks" OFF)

add_library(landing src/manager.cpp)
target_include_directories(landing PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(landing PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(LANDING_SANITIZERS)
    target_compile_options(landing PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(landing PUBLIC -fsanitize=address,undefined)
  endif()
endif()

add_executable(landing_demo apps/landing_demo.cpp)
target_link_libraries(landing_demo PRIVATE landing)

include(CTest)
if(BUILD_TESTING)
  add_executable(landing_tests tests/manager_test.cpp)
  target_link_libraries(landing_tests PRIVATE landing)
  foreach(case IN ITEMS normal recovery freshness duplicate out_of_order invalid
                        alignment disabled clock expiry initial_loss gap)
    add_test(NAME ${case} COMMAND landing_tests ${case})
  endforeach()
endif()
